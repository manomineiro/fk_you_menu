local function validBind(name)
	if type(name) ~= "string" then return false end
	if name == "MouseButton2" or name == "MouseButton3" then return true end
	local ok, v = pcall(function() return Enum.KeyCode[name] end)
	return ok and v ~= nil and v ~= Enum.KeyCode.Unknown
end
local function matches(input, bind)
	return input.KeyCode.Name == bind or input.UserInputType.Name == bind
end
local function isDown(bind)
	if bind == "MouseButton2" or bind == "MouseButton3" then
		return UserInputService:IsMouseButtonPressed(Enum.UserInputType[bind])
	end
	local ok, v = pcall(function() return UserInputService:IsKeyDown(Enum.KeyCode[bind]) end)
	return ok and v
end

------------------------------------------------------------------ Hitbox
local originalHitbox = {}
local function restoreHitbox(part)
	local o = originalHitbox[part]
	if not o or not part or not part.Parent then originalHitbox[part] = nil; return end
	pcall(function()
		part.Size = o.size; part.Transparency = o.transparency
		part.CanCollide = o.cancollide; if o.color then part.Color = o.color end
	end)
	originalHitbox[part] = nil
end
local function restoreAllHitboxes()
	for part in pairs(originalHitbox) do restoreHitbox(part) end
end
local function applyHitboxTo(player)
	if player == LocalPlayer then return end
	local char = player.Character; if not char then return end
	if CONFIG.AimTeamCheck and player.Team ~= nil and player.Team == LocalPlayer.Team then return end
	local part = char:FindFirstChild(CONFIG.HitboxPart)
	if not part or not part:IsA("BasePart") then return end
	if not originalHitbox[part] then
		originalHitbox[part] = { size=part.Size, transparency=part.Transparency,
			cancollide=part.CanCollide, color=part.Color }
	end
	part.Size = Vector3.new(CONFIG.HitboxSize, CONFIG.HitboxSize, CONFIG.HitboxSize)
	part.CanCollide = false
	if CONFIG.HitboxVisible then
		part.Transparency = CONFIG.HitboxTransparency; part.Color = ACCENT
	else part.Transparency = 1 end
end

local DEAD = { [Enum.HumanoidStateType.Dead]=true, [Enum.HumanoidStateType.Physics]=true }
local function isAlive(hum)
	if not hum or hum.Health <= 0 then return false end
	local ok, s = pcall(function() return hum:GetState() end)
	return not (ok and DEAD[s])
end
local function getColor(p)
	if CONFIG.UseTeamColor and p.Team then return p.TeamColor.Color end
	if p.Team ~= nil and p.Team == LocalPlayer.Team then return CONFIG.AllyColor end
	return CONFIG.EnemyColor
end
local RED, GREEN = Color3.new(1,0.15,0.15), Color3.new(0.2,1,0.3)
local function healthColor(hum)
	local f = hum.MaxHealth > 0 and math.clamp(hum.Health/hum.MaxHealth, 0, 1) or 0
	return RED:Lerp(GREEN, f)
end

local function hookPlayer(player)
	if player == LocalPlayer then return end
	local conns = {}; playerConns[player] = conns
	if player.Character then task.spawn(createEsp, player, player.Character) end
	table.insert(conns, player.CharacterAdded:Connect(function(c)
		createEsp(player, c); if refreshPlayerList then refreshPlayerList() end
	end))
	table.insert(conns, player.CharacterRemoving:Connect(function() destroyEsp(player) end))
	if refreshPlayerList then refreshPlayerList() end
end
local function unhookPlayer(player)
	destroyEsp(player)
	local c = playerConns[player]
	if c then for _, x in ipairs(c) do x:Disconnect() end; playerConns[player] = nil end
	if refreshPlayerList then refreshPlayerList() end
end
for _, p in ipairs(Players:GetPlayers()) do hookPlayer(p) end
track(Players.PlayerAdded:Connect(hookPlayer))
track(Players.PlayerRemoving:Connect(unhookPlayer))

local function isIgnored(p)
	return ignoreSet[p.Name:lower()] or ignoreSet[p.DisplayName:lower()] or false
end
rebuildIgnore = function()
	table.clear(ignoreSet)
	for name in string.gmatch(CONFIG.IgnoreList, "[^,;\n]+") do
		name = name:match("^%s*(.-)%s*$"):lower()
		if name ~= "" then ignoreSet[name] = true end
	end
end
local function hideEsp(d)
	d.highlight.Enabled = false; d.gui.Enabled = false
	d.tracer.Visible = false; d.box.Visible = false
	if d.predLine then d.predLine.Visible = false end
end
local function predictedAimPos(target, hum, predict)
	if predict <= 0 then return target.Position end
	local pos = target.Position + target.AssemblyLinearVelocity * predict
	if hum then
		local ok, s = pcall(function() return hum:GetState() end)
		if ok and (s == Enum.HumanoidStateType.Freefall or s == Enum.HumanoidStateType.Jumping) then
			local g = workspace.Gravity
			pos = pos + Vector3.new(0, -0.5*g*predict*predict, 0)
		end
	end
	return pos
end

			else

						local s = CONFIG.MaxDistance * 0.4

					end

					hl.FillColor, hl.OutlineColor = color, color
					hl.FillTransparency = CONFIG.FillTransparency + (1 - CONFIG.FillTransparency) * fade
					hl.OutlineTransparency = CONFIG.OutlineTransparency + (1 - CONFIG.OutlineTransparency) * fade
					hl.Enabled = true
					data.gui.Enabled = true
					data.label.TextColor3 = CONFIG.HealthColor and healthColor(hum) or color
					data.label.TextTransparency = fade
					data.label.TextStrokeTransparency = 0.3 + 0.7 * fade

					if CONFIG.ShowDistance then table.insert(parts, string.format("[%dm]", math.floor(dist))) end
					if CONFIG.ShowHealth then table.insert(parts, string.format("HP %d", math.floor(hum.Health))) end
					data.label.Text = table.concat(parts, " ")
					local showBox, showTracer, showPred = false, false, false
					if cam then
						local top = cam:WorldToViewportPoint(root.Position + Vector3.new(0, 3, 0))
						local bot = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3.2, 0))
						if top.Z > 0 and bot.Z > 0 then
							if CONFIG.ShowBox then
								local h = math.abs(bot.Y - top.Y); local w = h * 0.55
								data.box.Position = UDim2.fromOffset(top.X - w/2, math.min(top.Y, bot.Y))
								data.box.Size = UDim2.fromOffset(w, h)
								data.boxStroke.Color = color; data.boxStroke.Transparency = fade
								data.barBg.Visible = CONFIG.ShowHealthBar

									data.barFill.Size = UDim2.fromScale(1, f)
									data.barFill.BackgroundColor3 = healthColor(hum)
									data.barFill.BackgroundTransparency = fade
									data.barBg.BackgroundTransparency = 0.4 + 0.6 * fade
								end
								showBox = true
							end
							if CONFIG.ShowTracers then
								local vp = cam.ViewportSize; local from
								if CONFIG.TracerOrigin == "Center" then from = Vector2.new(vp.X/2, vp.Y/2)
								elseif CONFIG.TracerOrigin == "Cursor" then from = UserInputService:GetMouseLocation()
								else from = Vector2.new(vp.X/2, vp.Y) end
								local to = Vector2.new(top.X, (top.Y + bot.Y)/2)
								local delta = to - from

								data.tracer.Size = UDim2.fromOffset(delta.Magnitude, 1.5)
								data.tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))

										local delta = to - from

										data.predLine.Rotation = math.deg(math.atan2(delta.Y, delta.X))
										data.predLine.BackgroundColor3 = ACCENT2

		end

------------------------------------------------------------------ Silent Aim
local silentHooks = {}
local function getSilentTarget()
	local cam = workspace.CurrentCamera
	local myChar = LocalPlayer.Character
	if not cam or not myChar then return nil end
	local mousePos = UserInputService:GetMouseLocation()
	local best, bestScore
	for player, data in pairs(objects) do
		local char, hum = data.character, data.humanoid
		local part = char and resolveAimPart(char, CONFIG.SilentPart)
		if part and isAlive(hum)
			and not (CONFIG.SilentTeamCheck and player.Team ~= nil and player.Team == LocalPlayer.Team)
			and not isIgnored(player) then
			local pos, onScreen = cam:WorldToViewportPoint(part.Position)
			if onScreen and pos.Z > 0 then
				local sd = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
				if sd <= CONFIG.SilentFov then
					local blocked = false
					if CONFIG.SilentWallCheck then
						local params = RaycastParams.new()
						params.FilterType = Enum.RaycastFilterType.Exclude
						params.FilterDescendantsInstances = { myChar, char }
						params.IgnoreWater = true
						local hit = workspace:Raycast(cam.CFrame.Position,
							part.Position - cam.CFrame.Position, params)
						if hit and hit.Instance and not hit.Instance:IsDescendantOf(char) then
							blocked = true
						end
					end
					if not blocked and (not bestScore or sd < bestScore) then
						best, bestScore = part, sd
					end
				end
			end
		end
	end
	return best
end
local function installSilentHooks()
	if not hookmetamethod or not getnamecallmethod or not checkcaller then
		notify("Executor sem suporte a hookmetamethod", "err"); return false
	end
	if silentHooks.installed then return true end
	silentHooks.installed = true
	silentHooks.oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
		local method = getnamecallmethod()
		local args = { ... }
		if not checkcaller() and CONFIG.SilentEnabled and CONFIG.SilentHookRaycast then
			if (method == "Raycast" or method == "Spherecast" or method == "Blockcast")
				and self == workspace then
				local target = getSilentTarget()
				if target and args[1] and typeof(args[1]) == "Vector3" then
					args[2] = (target.Position - args[1])
					return silentHooks.oldNamecall(self, unpack(args))
				end
			elseif method == "FindPartOnRayWithIgnoreList"
				or method == "FindPartOnRay"
				or method == "FindPartOnRayWithWhitelist" then
				local target = getSilentTarget()
				if target and args[1] and typeof(args[1]) == "Ray" then
					args[1] = Ray.new(args[1].Origin, (target.Position - args[1].Origin))
					return silentHooks.oldNamecall(self, unpack(args))
				end
			end
		end
		return silentHooks.oldNamecall(self, ...)
	end)
	return true
end

local flyParts = { bv=nil, bg=nil, hum=nil }
local function clearFly()
	if flyParts.bv then flyParts.bv:Destroy(); flyParts.bv = nil end
	if flyParts.bg then flyParts.bg:Destroy(); flyParts.bg = nil end
	if flyParts.hum then pcall(function() flyParts.hum.PlatformStand = false end); flyParts.hum = nil end
end
local function startFly()
	clearFly()
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not (root and hum) then return end
	hum.PlatformStand = true
	local bv = Instance.new("BodyVelocity", root)
	bv.Name = "FlyBV"; bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
	bv.Velocity = Vector3.zero
	local bg = Instance.new("BodyGyro", root)
	bg.Name = "FlyBG"; bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
	bg.P = 9000; bg.CFrame = root.CFrame
	flyParts.bv, flyParts.bg, flyParts.hum = bv, bg, hum
end
local function flyStep()
	if not CONFIG.FlyEnabled then clearFly(); return end
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local cam = workspace.CurrentCamera
	if not (root and cam) then return end
	if not flyParts.bv or flyParts.bv.Parent ~= root then startFly() end
	if not flyParts.bv then return end
	local dir = Vector3.zero
	local f, r, u = cam.CFrame.LookVector, cam.CFrame.RightVector, Vector3.new(0,1,0)
	if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + f end
	if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - f end
	if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + r end
	if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - r end
	if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + u end
	if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - u end
	if dir.Magnitude > 0 then dir = dir.Unit end
	flyParts.bv.Velocity = dir * CONFIG.FlySpeed
	flyParts.bg.CFrame = cam.CFrame
end
local function applySpeed()
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	hum.WalkSpeed = CONFIG.SpeedEnabled and CONFIG.WalkSpeed or 16
	if CONFIG.JumpBoost then hum.UseJumpPower = true; hum.JumpPower = CONFIG.JumpPower
	else hum.JumpPower = 50 end
end
track(RunService.Stepped:Connect(function()

			for _, part in ipairs(char:GetDescendants()) do

					part.CanCollide = false
				end
			end

end))
track(RunService.RenderStepped:Connect(function()

end))

------------------------------------------------------------------ Teleportes, Waypoints e Spec
local function myRootPart()
	local char = LocalPlayer.Character
	return char and (char:FindFirstChild("HumanoidRootPart") or resolveAimPart(char, "HumanoidRootPart"))
end
local function savePos() local r = myRootPart(); if r then lastPosition = r.CFrame end end
local function teleportTo(cf)
	local r = myRootPart(); if not r then return false end
	savePos()
	if CONFIG.TpSmooth then
		local dist = (cf.Position - r.Position).Magnitude
		local time = math.clamp(dist / math.max(1, CONFIG.TpSmoothSpeed), 0.05, 3)
		TweenService:Create(r, TweenInfo.new(time, Enum.EasingStyle.Linear), { CFrame = cf }):Play()
	else r.CFrame = cf end
	return true
end
local function tpToPlayer(player, offset)
	if not player or not player.Character then return notify("Alvo inválido", "err") end
	local target = player.Character:FindFirstChild("HumanoidRootPart")
		or resolveAimPart(player.Character, "HumanoidRootPart")
	if not target then return notify("Alvo sem corpo", "err") end
	if teleportTo(target.CFrame * CFrame.new(0, 0, offset or CONFIG.FollowOffset)) then
		notify("TP → " .. player.DisplayName, "ok")
	end
end
local function tpToCursor()
	local cam = workspace.CurrentCamera
	local mouse = UserInputService:GetMouseLocation()
	if not cam then return end
	local ray = cam:ViewportPointToRay(mouse.X, mouse.Y)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { LocalPlayer.Character }
	params.IgnoreWater = true
	local hit = workspace:Raycast(ray.Origin, ray.Direction * 5000, params)
	if hit then
		if teleportTo(CFrame.new(hit.Position + Vector3.new(0, 3, 0))) then notify("TP cursor", "ok") end
	else notify("Raio não atingiu nenhum objeto", "err") end
end
local function tpBack()
	if not lastPosition then return notify("Sem posição anterior", "err") end
	local current = lastPosition; savePos()
	local r = myRootPart(); if r then r.CFrame = current; notify("Go Back", "ok") end
end
local function tpCoord(x,y,z)
	x,y,z = tonumber(x), tonumber(y), tonumber(z)
	if not (x and y and z) then return notify("Coordenadas inválidas", "err") end
	if teleportTo(CFrame.new(x,y,z)) then
		notify(string.format("TP → %.0f, %.0f, %.0f", x,y,z), "ok")
	end
end
local function saveWaypoint(name)
	name = name:match("^%s*(.-)%s*$")
	if name == "" then return notify("Nome vazio", "err") end
	local r = myRootPart(); if not r then return notify("Sem corpo", "err") end
	waypoints[name] = r.Position; notify("Waypoint salvo: " .. name, "ok")
	if refreshWaypoints then refreshWaypoints() end
end
local function loadWaypoint(name)
	local pos = waypoints[name]; if not pos then return notify("Waypoint não existe", "err") end
	if teleportTo(CFrame.new(pos)) then notify("TP → " .. name, "ok") end
end
local function deleteWaypoint(name)
	waypoints[name] = nil; notify("Removido: " .. name)
	if refreshWaypoints then refreshWaypoints() end
end

	if cam then
		local myChar = LocalPlayer.Character
		local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
		cam.CameraSubject = myHum or originalCamSubject
		cam.CameraType = Enum.CameraType.Custom
	end
	local n = spectating.DisplayName; spectating = nil; CONFIG.SpectateTarget = ""
	notify("Spec off (" .. n .. ")")
end

	local hum = player.Character:FindFirstChildOfClass("Humanoid")
	if not hum then return notify("Alvo sem humanoid", "err") end
	local cam = workspace.CurrentCamera; if not cam then return end
	if not spectating then originalCamSubject = cam.CameraSubject end
	spectating = player; CONFIG.SpectateTarget = player.Name
	cam.CameraSubject = hum; cam.CameraType = Enum.CameraType.Custom
	notify("Spec → " .. player.DisplayName, "ok")
end

------------------------------------------------------------------ Descarregamento (Unload)
local function unload()
	if unloaded then return end
	unloaded = true
	for _, c in ipairs(globalConns) do c:Disconnect() end
	table.clear(globalConns)
	for p in pairs(playerConns) do unhookPlayer(p) end
	for p in pairs(objects) do destroyEsp(p) end
	restoreAllHitboxes(); clearFly(); stopSpectate()
	local char = LocalPlayer.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if hum then hum.WalkSpeed, hum.JumpPower = 16, 50 end
	screenGui:Destroy()
	if env.__ESP_MENU_UNLOAD == unload then env.__ESP_MENU_UNLOAD = nil end
end
env.__ESP_MENU_UNLOAD = unload

------------------------------------------------------------------ Perfil e Configurações (JSON)
local function configFile() return "esp_menu_p" .. CONFIG.Profile .. ".json" end
local function refreshAll() for _, fn in ipairs(refreshers) do fn() end; refreshHud() end
local function saveConfig()
	if not writefile then return notify("Sem writefile no executor", "err") end
	local t = {}
	for _, k in ipairs(SAVE_KEYS) do t[k] = CONFIG[k] end
	for _, k in ipairs(COLOR_KEYS) do local c = CONFIG[k]; t[k] = { c.R, c.G, c.B } end
	t._waypoints = {}
	for n, pos in pairs(waypoints) do t._waypoints[n] = { pos.X, pos.Y, pos.Z } end
	local ok = pcall(function() writefile(configFile(), HttpService:JSONEncode(t)) end)
	notify(ok and ("Perfil " .. CONFIG.Profile .. " salvo") or "Falha ao salvar", ok and "ok" or "err")
end
local function loadConfig()
	if not (readfile and isfile) then return notify("Sem readfile no executor", "err") end
	local okE, exists = pcall(isfile, configFile())
	if not okE or not exists then return notify("Arquivo de perfil vazio", "err") end
	local ok, data = pcall(function() return HttpService:JSONDecode(readfile(configFile())) end)
	if not ok or type(data) ~= "table" then return notify("Configuração inválida", "err") end
	for _, k in ipairs(SAVE_KEYS) do
		local v = data[k]
		if v ~= nil and type(v) == type(CONFIG[k]) then
			local good = true
			if BIND_KEYS[k] then good = validBind(v) end
			if choices[k] and not choices[k][v] then good = false end
			if good then CONFIG[k] = v end
		end
	end
	for _, k in ipairs(COLOR_KEYS) do
		local c = data[k]
		if type(c) == "table" and type(c[1]) == "number" then
			CONFIG[k] = Color3.new(math.clamp(c[1],0,1), math.clamp(c[2],0,1), math.clamp(c[3],0,1))
		end
	end
	if type(data._waypoints) == "table" then
		table.clear(waypoints)
		for n, v in pairs(data._waypoints) do
			if type(v) == "table" and type(v[1]) == "number" then
				waypoints[n] = Vector3.new(v[1], v[2], v[3])
			end
		end
	end
	aimToggled = false; rebuildIgnore(); applyTheme(); applySpeed(); applyUiMode()
	if refreshWaypoints then refreshWaypoints() end
	notify("Perfil " .. CONFIG.Profile .. " carregado", "ok")
end

			function(x)

				if ch == "R" then r=x elseif ch == "G" then g=x else b=x end

local function addCycle(page, text, key, options, onChange)
	local row = newRow(page, 32); addLabel(row, text, 6)

	local function refresh()
		for _, o in ipairs(options) do if o[1] == CONFIG[key] then v.Text = o[2] .. "  ›" end end
	end
	refresh(); table.insert(refreshers, refresh)
	local hit = Instance.new("TextButton", row)
	hit.Size = UDim2.fromScale(1,1); hit.BackgroundTransparency = 1; hit.Text = ""
	track(hit.MouseButton1Click:Connect(function()
		for i, o in ipairs(options) do
			if o[1] == CONFIG[key] then CONFIG[key] = options[i % #options + 1][1]; break end
		end
		refresh(); refreshHud(); if onChange then onChange(CONFIG[key]) end
	end))
end

applyTheme = function()
	local pair = THEMES[CONFIG.Theme] or THEMES.Roxo
	ACCENT, ACCENT2 = pair[1], pair[2]
	for i = #accentObjs, 1, -1 do
		local o = accentObjs[i]
		if o[1].Parent then
			if o[2] == "_gradient" then
				o[1].Color = ColorSequence.new{
					ColorSequenceKeypoint.new(0, ACCENT), ColorSequenceKeypoint.new(1, ACCENT2) }
			else o[1][o[2]] = ACCENT end
		else table.remove(accentObjs, i) end
	end
	refreshAll(); selectTab(currentTab)
end

	end))

local function getHumanizedTarget(cam, mousePos, myChar)
	local best, bestScore
	for player, data in pairs(objects) do
		local char, hum = data.character, data.humanoid
		local part = char and resolveAimPart(char, CONFIG.AimPart)
		if part and isAlive(hum)
			and not (CONFIG.AimTeamCheck and player.Team ~= nil and player.Team == LocalPlayer.Team)
			and not isIgnored(player) then
			local pos, onScreen = cam:WorldToViewportPoint(part.Position)
			if onScreen and pos.Z > 0 then
				local sd = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
				if sd <= CONFIG.AimFov then
					local score
					if CONFIG.AimPriority == "Distance" then
						score = (part.Position - cam.CFrame.Position).Magnitude
					elseif CONFIG.AimPriority == "Health" then score = hum.Health
					else score = sd end
					if not bestScore or score < bestScore then
						local blocked = false
						if CONFIG.AimWallCheck then
							local params = RaycastParams.new()
							params.FilterType = Enum.RaycastFilterType.Exclude
							params.FilterDescendantsInstances = { myChar, char }
							params.IgnoreWater = true
							local hit = workspace:Raycast(cam.CFrame.Position,
								part.Position - cam.CFrame.Position, params)
							if hit and hit.Instance and not hit.Instance:IsDescendantOf(char) then
								blocked = true
							end
						end
						if not blocked then best, bestScore = part, score end
					end
				end
			end
		end
	end
	return best
end

local aimSeed = math.random(1, 10000)

track(RunService.RenderStepped:Connect(function(dt)
	local mousePos = UserInputService:GetMouseLocation()
	fovCircle.Visible = CONFIG.ShowFov and CONFIG.AimEnabled
	fovCircle.Position = UDim2.fromOffset(mousePos.X, mousePos.Y)
	fovCircle.Size = UDim2.fromOffset(CONFIG.AimFov*2, CONFIG.AimFov*2)
	fovStroke.Color = Color3.new(1,1,1)
	
	if not CONFIG.AimEnabled then return end
	local active
	if CONFIG.AimMode == "Toggle" then active = aimToggled
	else active = (not listening) and isDown(CONFIG.AimKey) end
	if not active then return end

	local cam = workspace.CurrentCamera
	local myChar = LocalPlayer.Character
	if not cam or not myChar then return end

	local target = getHumanizedTarget(cam, mousePos, myChar)
	if not target then return end

	fovStroke.Color = Color3.fromRGB(255,80,80)
	local hum = target.Parent and target.Parent:FindFirstChildOfClass("Humanoid")
	local aimPos = predictedAimPos(target, hum, CONFIG.AimPredict)

	if CONFIG.HumanizeAim then
		aimSeed += dt * 5
		local noiseX = math.noise(aimSeed, 0, 0) * (CONFIG.HumanizeJitter or 1.2)
		local noiseY = math.noise(0, aimSeed, 0) * (CONFIG.HumanizeJitter or 1.2)
		local distFactor = (aimPos - cam.CFrame.Position).Magnitude
		local arcOffset = Vector3.new(noiseX, noiseY + (distFactor * 0.002 * (CONFIG.HumanizeCurvature or 0.35)), noiseX)
		aimPos = aimPos + arcOffset
	end

	local smoothness = math.max(1, CONFIG.AimSmooth)
	local alpha = 1 - (1 - 1/smoothness)^(dt * 60)
	local goal = CFrame.new(cam.CFrame.Position, aimPos)
	cam.CFrame = cam.CFrame:Lerp(goal, alpha)
end))

track(UserInputService.InputBegan:Connect(function(input, processed)
	if listening then
		local name
		if input.UserInputType == Enum.UserInputType.Keyboard then name = input.KeyCode.Name
		elseif input.UserInputType == Enum.UserInputType.MouseButton2
			or input.UserInputType == Enum.UserInputType.MouseButton3 then
			name = input.UserInputType.Name
		end
		if name then
			local l = listening; listening = nil
			if name ~= "Escape" and validBind(name) then
				CONFIG[l.key] = name; notify("Tecla alterada: " .. name, "ok")
			end
			l.refresh()
		end
		return
	end
	if processed then return end
	if matches(input, CONFIG.ToggleKey) then
		CONFIG.Enabled = not CONFIG.Enabled; refreshEnabled(true); refreshHud()
		notify(CONFIG.Enabled and "ESP ligado" or "ESP desligado")
	elseif matches(input, CONFIG.MenuKey) then
		if CONFIG.UiMode == "Full" then setMenu(not menuShown)
		else hud.Visible = not hud.Visible end
	elseif matches(input, CONFIG.UiModeKey) then
		CONFIG.UiMode = (CONFIG.UiMode == "Full") and "Compact" or "Full"
		applyUiMode(); refreshAll()
		notify("Modo UI: " .. (CONFIG.UiMode == "Full" and "Janela" or "Mini-HUD"), "ok")

		CONFIG.FlyEnabled = not CONFIG.FlyEnabled
		if CONFIG.FlyEnabled then startFly() else clearFly() end

		CONFIG.NoclipEnabled = not CONFIG.NoclipEnabled
		refreshAll(); notify(CONFIG.NoclipEnabled and "Noclip on" or "Noclip off")
	elseif matches(input, CONFIG.TpCursorKey) then tpToCursor()
	elseif matches(input, CONFIG.TpBackKey) then tpBack() end
	if CONFIG.AimMode == "Toggle" and CONFIG.AimEnabled and matches(input, CONFIG.AimKey) then
